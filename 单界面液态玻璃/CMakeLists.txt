cmake_minimum_required(VERSION 3.24)
project(QtLiquidSurface VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
if(APPLE)
    enable_language(OBJCXX)
    set(CMAKE_OSX_DEPLOYMENT_TARGET 26.0)
endif()
find_package(Qt6 6.10 REQUIRED COMPONENTS Quick ShaderTools)
qt_standard_project_setup()
qt_add_executable(QtLiquidSurface WIN32 MACOSX_BUNDLE main.cpp)
target_link_libraries(QtLiquidSurface PRIVATE Qt6::Quick)
if(APPLE)
    target_sources(QtLiquidSurface PRIVATE Accessibility.mm)
    target_link_libraries(QtLiquidSurface PRIVATE "-framework AppKit")
    set_source_files_properties(Accessibility.mm PROPERTIES COMPILE_OPTIONS "-fobjc-arc")
else()
    target_sources(QtLiquidSurface PRIVATE Accessibility.cpp)
    if(WIN32)
        target_link_libraries(QtLiquidSurface PRIVATE user32 advapi32)
        target_compile_definitions(QtLiquidSurface PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
    endif()
endif()
if(MSVC)
    target_compile_options(QtLiquidSurface PRIVATE /W4 /WX /utf-8)
else()
    target_compile_options(QtLiquidSurface PRIVATE -Wall -Wextra -Werror)
endif()
set_target_properties(QtLiquidSurface PROPERTIES
    MACOSX_BUNDLE_GUI_IDENTIFIER local.faunabox.qt-liquid-surface
    MACOSX_BUNDLE_BUNDLE_NAME "Qt Liquid Surface"
    MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/Info.plist")
qt_add_resources(QtLiquidSurface scene PREFIX "/" FILES Main.qml)
qt_add_shaders(QtLiquidSurface optics PREFIX "/shaders" BASE "${CMAKE_CURRENT_SOURCE_DIR}"
    FILES background.frag glass.frag)
enable_testing()
add_test(NAME qt-liquid-surface-gpu COMMAND QtLiquidSurface --self-test)
set_tests_properties(qt-liquid-surface-gpu PROPERTIES TIMEOUT 30)
